Reactive forms

import { Component, inject, output } from '@angular/core';
import { FormBuilder, ReactiveFormsModule, Validators } from '@angular/forms';

@Component({
  selector: 'app-assignment-form',
  imports: [ReactiveFormsModule],
  templateUrl: './assignment-form.html',
})
export class AssignmentForm {
  private fb = inject(FormBuilder).nonNullable;   // nonNullable: reset() về giá trị đầu chứ không về null
  saved = output();

  form = this.fb.group({
    title: ['', [Validators.required, Validators.maxLength(200)]],
    subject: ['toan', Validators.required],
    dueDate: ['', Validators.required],
    exerciseCount: [1, [Validators.required, Validators.min(1), Validators.max(20)]],
    note: ['', Validators.maxLength(500)],
  });

  submit() {
    if (this.form.invalid) {
      this.form.markAllAsTouched();     // hiện lỗi ở mọi ô
      return;
    }
    this.saved.emit(this.form.getRawValue()); // kiểu: { title: string; subject: string; ... }
    this.form.reset();
  }
}

Validator tự viết

import { AbstractControl, ValidationErrors, ValidatorFn } from '@angular/forms';

// Không cho chọn ngày trong quá khứ
export function notInPast(today = () => new Date().toISOString().slice(0, 10)): ValidatorFn {
  return (c: AbstractControl): ValidationErrors | null =>
    c.value && c.value < today() ? { inPast: true } : null;
}

// Validator cho cả nhóm: ngày kết thúc phải sau ngày bắt đầu
export const rangeOrder: ValidatorFn = (g: AbstractControl) => {
  const { start, end } = g.value as { start: string; end: string };
  return start && end && end < start ? { rangeOrder: true } : null;
};

form = this.fb.group(
  {
    start: ['', Validators.required],
    end: ['', [Validators.required, notInPast()]],
  },
  { validators: rangeOrder },   // validator cấp nhóm
);

export function classNameAvailable(api: ClassApi): AsyncValidatorFn {
  return (c) =>
    timer(400).pipe(                                   // chờ người dùng gõ xong (debounce)

export class QuizForm {

  form = this.fb.group({

Hiển thị lỗi gọn: một component dùng lại

const MESSAGES: Record string> = {
  required: () => 'Không được để trống',
  maxlength: (e) => `Tối đa ${e.requiredLength} ký tự`,
  min: (e) => `Tối thiểu ${e.min}`,
  max: (e) => `Tối đa ${e.max}`,
  email: () => 'Email không đúng định dạng',
  inPast: () => 'Không chọn ngày đã qua',
  taken: () => 'Tên đã được dùng',
};

@Component({
  selector: 'app-field-error',
  template: `
    @if (control().invalid && (control().touched || control().dirty)) {
      {{ message() }}
    }
  `,
})
export class FieldError {
  control = input.required();
  protected message() {
    const errs = this.control().errors ?? {};
    const key = Object.keys(errs)[0];
    return key ? (MESSAGES[key]?.(errs[key]) ?? 'Không hợp lệ') : '';
  }
}

message() là method nên được gọi lại mỗi lần Angular cập nhật component: gõ, blur, bấm lưu đều kích hoạt cập nhật nên lỗi luôn đúng. Trạng thái của form (touched, errors) chưa phải signal; ứng dụng zoneless vẫn cập nhật vì sự kiện DOM trong template luôn đánh dấu component cần kiểm tra lại. Nếu lỗi được đặt từ code bên ngoài (ví dụ lỗi từ server), gọi control.setErrors(...) trong lúc xử lý sự kiện hoặc kèm một signal để giao diện biết.

Signal forms: hướng đi mới

Từ Angular 21, có thêm API signal forms (gói @angular/forms/signals, đang ở trạng thái thử nghiệm). Model của form chính là một signal, quy tắc validate khai báo bằng schema:

import { form, Field, required, maxLength, min } from '@angular/forms/signals';

@Component({
  imports: [Field],
  template: `
    
    @if (f.title().invalid() && f.title().touched()) { {{ f.title().errors()[0].message }} }
    
  `,
})
export class AssignmentFormSignal {
  model = signal({ title: '', exerciseCount: 1 });
  f = form(this.model, (p) => {
    required(p.title, { message: 'Nhập tên bài' });
    maxLength(p.title, 200);
    min(p.exerciseCount, 1);
  });
}

Signal forms còn ở giai đoạn thử nghiệm: tên hàm và cách dùng có thể đổi giữa các phiên bản (ví dụ directive ô nhập đã đổi tên trong quá trình phát triển). Xem tài liệu ứng với phiên bản trong dự án. Dự án production hiện vẫn nên dùng reactive forms.

Signal forms rất giống tinh thần Vue: model là state thường (ref), form chỉ là lớp kiểm tra bao quanh. Nếu bạn quen VeeValidate với schema Zod, cách khai báo này sẽ thấy tự nhiên.

Ô nhập tự viết dùng được với forms

Component tự viết (chọn ngày, chọn màu, rating) muốn dùng formControlName phải cài đặt ControlValueAccessor:

@Component({
  selector: 'app-star-input',
  providers: [{ provide: NG_VALUE_ACCESSOR, useExisting: forwardRef(() => StarInput), multi: true }],
  template: `
    @for (n of [1, 2, 3, 4, 5]; track n) {
      
    }
  `,
})
export class StarInput implements ControlValueAccessor {
  protected value = signal(0);
  protected disabled = signal(false);
  private onChange: (v: number) => void = () => {};
  private onTouched: () => void = () => {};

  writeValue(v: number | null) { this.value.set(v ?? 0); }        // form → component
  registerOnChange(fn: (v: number) => void) { this.onChange = fn; }
  registerOnTouched(fn: () => void) { this.onTouched = fn; }
  setDisabledState(d: boolean) { this.disabled.set(d); }

  protected pick(n: number) {
    this.value.set(n);
    this.onChange(n);                                             // component → form
    this.onTouched();
  }
}

Đây là thứ Vue làm rất gọn bằng defineModel. Ở Angular dài hơn nhưng đổi lại component dùng được với mọi loại form (template-driven, reactive). Với signal forms, chỉ cần model() là đủ.

- control.invalid + control.invalid && control.touched, và gọi form.markAllAsTouched() khi bấm lưu mà form không hợp lệ - control.dirty ? touched đúng sau khi blur. markAllAsTouched khiến mọi ô hiện lỗi khi người dùng bấm lưu sớm. - Có + Không, chỉ có trong form.getRawValue() - Có nhưng là null ? Control disabled bị loại khỏi value. Gửi API thường dùng getRawValue() nếu cần cả trường disabled.

Bài tập

Form đăng ký lớp có kiểm tra năm học và mật khẩu xác nhận

Form tạo lớp gồm: tên lớp (bắt buộc, tối đa 60 ký tự), sĩ số (1 đến 80), năm học dạng 2026-2027 (năm sau = năm trước + 1), mã tham gia và nhập lại mã (phải khớp). Nút lưu chỉ bật khi hợp lệ. Bấm lưu thì in giá trị ra console và reset form.

Gợi ý từng bước

  1. Năm học: validator hàm tách chuỗi bằng split('-'), kiểm tra 4 chữ số và y2 === y1 + 1.
  2. Mã khớp: validator cấp nhóm so code và code2, trả { codeMismatch: true }.
  3. Lỗi cấp nhóm hiển thị bằng form.hasError('codeMismatch'), chỉ khi ô code2 đã touched.
  4. Nút: [disabled]="form.invalid".
Xem lời giải 
import { Component, inject } from '@angular/core';
import { AbstractControl, FormBuilder, ReactiveFormsModule, ValidationErrors, Validators } from '@angular/forms';

function schoolYear(c: AbstractControl): ValidationErrors | null {
  const m = /^(\d{4})-(\d{4})$/.exec(c.value ?? '');
  return m && Number(m[2]) === Number(m[1]) + 1 ? null : { schoolYear: true };
}

function codesMatch(g: AbstractControl): ValidationErrors | null {
  const { code, code2 } = g.value as { code: string; code2: string };
  return code && code2 && code !== code2 ? { codeMismatch: true } : null;
}

    console.log('Tạo lớp', data);

}

Giải thích: lỗi khớp mã nằm ở nhóm (không phải ở ô code2) nên kiểm tra bằng form.hasError. Vì vậy ô code2 không tự đỏ viền: nếu muốn, thêm [class.ng-invalid]="form.hasError('codeMismatch')". Destructuring { code2, ...data } bỏ trường nhập lại trước khi gửi đi. nonNullable khiến reset() trả sĩ số về 30 và năm học về giá trị đầu, không về null.
