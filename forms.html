Forms: reactive forms và signal forms

Template-driven vs reactive forms, typed forms, validator tự viết và bất đồng bộ, FormArray, hiển thị lỗi, signal forms. So với v-model + VeeValidate.

Học xong bài này bạn sẽ:
  • Xây form phức tạp bằng reactive forms có kiểu
  • Viết validator đồng bộ và bất đồng bộ
  • Biết hướng đi signal forms của Angular

#Ba cách làm form trong Angular

Template-driven (ngModel)Reactive formsSignal forms
State nằm ởtemplateclass (FormGroup)signal chứa model
Giống Vuev-model từng ôVeeValidate / FormKit với schemav-model + validate theo schema

Hợp vớiform 2–3 ô đơn giảnmặc định cho form thật: nhiều trường, validator, trường độnghướng đi mới (thử nghiệm từ v21), nên theo dõi

Bài này tập trung vào reactive forms vì đó là thứ bạn gặp ở 90% dự án, rồi giới thiệu signal forms.

#Template-driven: nhanh cho form nhỏ

  imports: [FormsModule],

      @if (qCtrl.invalid && qCtrl.touched) { <small>Nhập ít nhất 2 ký tự</small> }

Validator viết như thuộc tính HTML (required, minlength, email, pattern). Gọn nhưng logic nằm rải trong template, khó test, khó làm form động. Dùng cho ô tìm kiếm, bộ lọc.

#Reactive forms

assignment-form.ts
import { Component, inject, output } from '@angular/core';
import { FormBuilder, ReactiveFormsModule, Validators } from '@angular/forms';

@Component({
  selector: 'app-assignment-form',
  imports: [ReactiveFormsModule],
  templateUrl: './assignment-form.html',
})
export class AssignmentForm {
  private fb = inject(FormBuilder).nonNullable;   // nonNullable: reset() về giá trị đầu chứ không về null
  saved = output<NewAssignment>();

  form = this.fb.group({
    title: ['', [Validators.required, Validators.maxLength(200)]],
    subject: ['toan', Validators.required],
    dueDate: ['', Validators.required],
    exerciseCount: [1, [Validators.required, Validators.min(1), Validators.max(20)]],
    note: ['', Validators.maxLength(500)],
  });

  submit() {
    if (this.form.invalid) {
      this.form.markAllAsTouched();     // hiện lỗi ở mọi ô
      return;
    }
    this.saved.emit(this.form.getRawValue()); // kiểu: { title: string; subject: string; ... }
    this.form.reset();
  }
}

Đọc form như signal

Reactive forms vẫn dựa trên Observable. Chuyển sang signal để dùng với computed:

TypeScript
private value = toSignal(this.form.valueChanges, { initialValue: this.form.getRawValue() });
preview = computed(() => `${this.value().title} — hạn ${this.value().dueDate}`);

#Validator tự viết

shared/validators.ts
import { AbstractControl, ValidationErrors, ValidatorFn } from '@angular/forms';

// Không cho chọn ngày trong quá khứ
export function notInPast(today = () => new Date().toISOString().slice(0, 10)): ValidatorFn {
  return (c: AbstractControl): ValidationErrors | null =>
    c.value && c.value < today() ? { inPast: true } : null;
}

// Validator cho cả nhóm: ngày kết thúc phải sau ngày bắt đầu
export const rangeOrder: ValidatorFn = (g: AbstractControl) => {
  const { start, end } = g.value as { start: string; end: string };
  return start && end && end < start ? { rangeOrder: true } : null;
};
TypeScript
form = this.fb.group(

Validator bất đồng bộ

TypeScript
export function classNameAvailable(api: ClassApi): AsyncValidatorFn {
  return (c) =>
    timer(400).pipe(                                   // chờ người dùng gõ xong (debounce)
      switchMap(() => api.nameExists(c.value)),
      map((exists) => (exists ? { taken: true } : null)),
      catchError(() => of(null)),                      // lỗi mạng: không chặn người dùng
    );
}

name = this.fb.control('', {
  validators: [Validators.required],
  asyncValidators: [classNameAvailable(inject(ClassApi))],
  updateOn: 'blur',                                    // chỉ kiểm tra khi rời ô
});

#Form động: FormArray

Giáo viên giao bài có nhiều câu hỏi, bấm "Thêm câu" để thêm dòng:

quiz-form.ts
export class QuizForm {
  private fb = inject(FormBuilder).nonNullable;

  form = this.fb.group({
    title: ['', Validators.required],
    questions: this.fb.array([this.newQuestion()]),
  });

  get questions() {
    return this.form.controls.questions;
  }

  newQuestion() {
    return this.fb.group({
      text: ['', Validators.required],
      points: [1, [Validators.required, Validators.min(0.5)]],
    });
  }

  add() { this.questions.push(this.newQuestion()); }
  remove(i: number) { this.questions.removeAt(i); }

  total = toSignal(this.questions.valueChanges.pipe(map((qs) => qs.reduce((t, q) => t + (q.points ?? 0), 0))), { initialValue: 1 });
}
Template
<form [formGroup]="form">
  <input formControlName="title" placeholder="Tên bài kiểm tra" />
  <div formArrayName="questions">
    @for (q of questions.controls; track q; let i = $index) {
      <div [formGroupName]="i" class="row">
        <span>Câu {{ i + 1 }}</span>
        <input formControlName="text" placeholder="Nội dung câu hỏi" />
        <input type="number" formControlName="points" step="0.5" />
        <button type="button" (click)="remove(i)" [disabled]="questions.length === 1">Xoá</button>
      </div>
    }
  </div>
  <button type="button" (click)="add()">+ Thêm câu</button>

#Điền dữ liệu, disable, lắng nghe

TypeScript
// Mở form sửa: điền dữ liệu đang có
this.form.patchValue({ title: a.title, dueDate: a.dueDate }); // patch: một phần; setValue: phải đủ mọi trường

// Bật / tắt theo điều kiện
this.form.controls.dueDate.disable();     // giá trị không nằm trong .value nữa, vẫn có trong getRawValue()

// Phản ứng khi một ô đổi
this.form.controls.subject.valueChanges.pipe(takeUntilDestroyed()).subscribe((s) => {
  if (s === 'tieng-viet') this.form.controls.exerciseCount.setValue(1);
});

#Hiển thị lỗi gọn: một component dùng lại

shared/field-error.ts
const MESSAGES: Record<string, (e: any) => string> = {
  required: () => 'Không được để trống',
  maxlength: (e) => `Tối đa ${e.requiredLength} ký tự`,
  min: (e) => `Tối thiểu ${e.min}`,
  max: (e) => `Tối đa ${e.max}`,
  email: () => 'Email không đúng định dạng',
  inPast: () => 'Không chọn ngày đã qua',
  taken: () => 'Tên đã được dùng',
};

@Component({
  selector: 'app-field-error',
  template: `
    @if (control().invalid && (control().touched || control().dirty)) {
      <small class="err">{{ message() }}</small>
    }
  `,
})
export class FieldError {
  control = input.required<AbstractControl>();
  protected message() {
    const errs = this.control().errors ?? {};
    const key = Object.keys(errs)[0];
    return key ? (MESSAGES[key]?.(errs[key]) ?? 'Không hợp lệ') : '';
  }
}
Template
<input formControlName="title" />
<app-field-error [control]="form.controls.title" />

message() là method nên được gọi lại mỗi lần Angular cập nhật component: gõ, blur, bấm lưu đều kích hoạt cập nhật nên lỗi luôn đúng. Trạng thái của form (touched, errors) chưa phải signal; ứng dụng zoneless vẫn cập nhật vì sự kiện DOM trong template luôn đánh dấu component cần kiểm tra lại. Nếu lỗi được đặt từ code bên ngoài (ví dụ lỗi từ server), gọi control.setErrors(...) trong lúc xử lý sự kiện hoặc kèm một signal để giao diện biết.

#Signal forms: hướng đi mới

Từ Angular 21, có thêm API signal forms (gói @angular/forms/signals, đang ở trạng thái thử nghiệm). Model của form chính là một signal, quy tắc validate khai báo bằng schema:

TypeScript
import { form, Field, required, maxLength, min } from '@angular/forms/signals';

@Component({
  imports: [Field],
  template: `
    <input [field]="f.title" />
    @if (f.title().invalid() && f.title().touched()) { <small>{{ f.title().errors()[0].message }}</small> }
    <input type="number" [field]="f.exerciseCount" />
  `,
})
export class AssignmentFormSignal {
  model = signal({ title: '', exerciseCount: 1 });
  f = form(this.model, (p) => {
    required(p.title, { message: 'Nhập tên bài' });
    maxLength(p.title, 200);
    min(p.exerciseCount, 1);
  });
}

Signal forms còn ở giai đoạn thử nghiệm: tên hàm và cách dùng có thể đổi giữa các phiên bản (ví dụ directive ô nhập đã đổi tên trong quá trình phát triển). Xem tài liệu ứng với phiên bản trong dự án. Dự án production hiện vẫn nên dùng reactive forms.

Signal forms rất giống tinh thần Vue: model là state thường (ref), form chỉ là lớp kiểm tra bao quanh. Nếu bạn quen VeeValidate với schema Zod, cách khai báo này sẽ thấy tự nhiên.

#Ô nhập tự viết dùng được với forms

Component tự viết (chọn ngày, chọn màu, rating) muốn dùng formControlName phải cài đặt ControlValueAccessor:

shared/star-input.ts
@Component({
  selector: 'app-star-input',
  providers: [{ provide: NG_VALUE_ACCESSOR, useExisting: forwardRef(() => StarInput), multi: true }],
  template: `
    @for (n of [1, 2, 3, 4, 5]; track n) {
      <button type="button" [class.on]="n <= value()" [disabled]="disabled()" (click)="pick(n)">★</button>
    }
  `,
})
export class StarInput implements ControlValueAccessor {
  protected value = signal(0);
  protected disabled = signal(false);
  private onChange: (v: number) => void = () => {};
  private onTouched: () => void = () => {};

  writeValue(v: number | null) { this.value.set(v ?? 0); }        // form → component
  registerOnChange(fn: (v: number) => void) { this.onChange = fn; }
  registerOnTouched(fn: () => void) { this.onTouched = fn; }
  setDisabledState(d: boolean) { this.disabled.set(d); }

  protected pick(n: number) {
    this.value.set(n);
    this.onChange(n);                                             // component → form
    this.onTouched();
  }
}
Template
<app-star-input formControlName="quality" />

Đây là thứ Vue làm rất gọn bằng defineModel. Ở Angular dài hơn nhưng đổi lại component dùng được với mọi loại form (template-driven, reactive). Với signal forms, chỉ cần model() là đủ.

#Bài tập

Form đăng ký lớp có kiểm tra năm học và mật khẩu xác nhận

Form tạo lớp gồm: tên lớp (bắt buộc, tối đa 60 ký tự), sĩ số (1 đến 80), năm học dạng 2026-2027 (năm sau = năm trước + 1), mã tham gia và nhập lại mã (phải khớp). Nút lưu chỉ bật khi hợp lệ. Bấm lưu thì in giá trị ra console và reset form.

Gợi ý từng bước

  1. Năm học: validator hàm tách chuỗi bằng split('-'), kiểm tra 4 chữ số và y2 === y1 + 1.
  2. Mã khớp: validator cấp nhóm so code và code2, trả { codeMismatch: true }.
  3. Lỗi cấp nhóm hiển thị bằng form.hasError('codeMismatch'), chỉ khi ô code2 đã touched.
  4. Nút: [disabled]="form.invalid".
Xem lời giải
class-create-form.ts
import { Component, inject } from '@angular/core';
import { AbstractControl, FormBuilder, ReactiveFormsModule, ValidationErrors, Validators } from '@angular/forms';

function schoolYear(c: AbstractControl): ValidationErrors | null {
  const m = /^(\d{4})-(\d{4})$/.exec(c.value ?? '');
  return m && Number(m[2]) === Number(m[1]) + 1 ? null : { schoolYear: true };
}

function codesMatch(g: AbstractControl): ValidationErrors | null {
  const { code, code2 } = g.value as { code: string; code2: string };
  return code && code2 && code !== code2 ? { codeMismatch: true } : null;
}

@Component({
  selector: 'app-class-create-form',
  imports: [ReactiveFormsModule],
  template: `
    <form [formGroup]="form" (ngSubmit)="save()">
      <label>Tên lớp <input formControlName="name" maxlength="60" /></label>
      @if (c.name.touched && c.name.invalid) { <small class="err">Nhập tên lớp (tối đa 60 ký tự)</small> }

      <label>Sĩ số <input type="number" formControlName="capacity" /></label>
      @if (c.capacity.touched && c.capacity.invalid) { <small class="err">Sĩ số từ 1 đến 80</small> }

      <label>Năm học <input formControlName="schoolYear" placeholder="2026-2027" /></label>
      @if (c.schoolYear.touched && c.schoolYear.hasError('schoolYear')) { <small class="err">Năm học dạng 2026-2027</small> }

      <label>Mã tham gia <input formControlName="code" /></label>
      <label>Nhập lại mã <input formControlName="code2" /></label>
      @if (c.code2.touched && form.hasError('codeMismatch')) { <small class="err">Hai mã không khớp</small> }

      <button [disabled]="form.invalid">Tạo lớp</button>
    </form>
  `,
  styles: `form { display: grid; gap: 6px; max-width: 360px; } .err { color: #dc2626; }`,
})
export class ClassCreateForm {
  private fb = inject(FormBuilder).nonNullable;
  form = this.fb.group(
    {
      name: ['', [Validators.required, Validators.maxLength(60)]],
      capacity: [30, [Validators.required, Validators.min(1), Validators.max(80)]],
      schoolYear: ['2026-2027', [Validators.required, schoolYear]],
      code: ['', Validators.required],
      code2: ['', Validators.required],
    },
    { validators: codesMatch },
  );
  protected c = this.form.controls;

  save() {
    const { code2, ...data } = this.form.getRawValue();
    console.log('Tạo lớp', data);
    this.form.reset();
  }
}
